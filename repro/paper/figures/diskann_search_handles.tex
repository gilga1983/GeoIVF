% Feedback routing plan (no changes to nodes or forward paths):
%   source = completed neighbor insertion (insert.west)
%   target = bounded search frontier (queue.east)
%   west lane is outside the leftmost lower-row node;
%   return lane stays below the SSD-read and PQ-scoring nodes;
%   east lane returns outside the graph-record box.
% The three dedicated orthogonal corridors keep the feedback arrow
% disjoint from query execution and the NavHints start interface.
\coordinate (feedback-west) at ($(insert.west)+(-4mm,0)$);
\coordinate (feedback-low) at ($(read.south)+(0,-9mm)$);
\coordinate (feedback-east) at ($(read.east)+(9mm,0)$);
\draw[looparr] (insert.west)
  -- (feedback-west |- insert.west)
  -- (feedback-west |- feedback-low)
  -- (feedback-east |- feedback-low)
  -- (feedback-east |- queue.east)
  -- (queue.east);

\node[font=\tiny, align=center, text=nhHandleMemory!85!black, below=1mm of pq] {resident memory};
\end{tikzpicture}
